    <!-- Interaction Logic -->

    <script>

        const booths = document.querySelectorAll('.booth');

        const rows = document.querySelectorAll('#exhibitor-list tr');


        // Clear all active states

        function clearActive() {

            booths.forEach(b => b.classList.remove('active'));

            rows.forEach(r => r.classList.remove('active'));

        }


        // Map Click -> Highlights Table Row

        booths.forEach(booth => {

            booth.addEventListener('click', () => {

                clearActive();

                booth.classList.add('active');

                

                const boothId = booth.getAttribute('data-booth');

                const targetRow = Array.from(rows).find(row => 

                    row.getAttribute('data-booths').includes(boothId)

                );


                if (targetRow) {

                    targetRow.classList.add('active');

                    targetRow.scrollIntoView({ behavior: 'smooth', block: 'center' });

                }

            });

        });


        // Table Row Click -> Highlights Map Booth

        rows.forEach(row => {

            row.addEventListener('click', (e) => {

                // Prevent overriding if they click the actual external link

                if(e.target.tagName.toLowerCase() === 'a') return; 


                clearActive();

                row.classList.add('active');

                

                const boothIds = row.getAttribute('data-booths').split(', ');

                boothIds.forEach(id => {

                    const targetBooth = document.querySelector(`.booth[data-booth="${id}"]`);

                    if (targetBooth) targetBooth.classList.add('active');

                });

            });

        });

    </script>
